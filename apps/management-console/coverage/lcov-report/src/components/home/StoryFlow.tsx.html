                            if (e.nativeEvent.isComposing) return;
                            if (showSlash && filteredCmds.length > 0) {
                                if (e.key === 'ArrowDown') {
                                    e.preventDefault();
                                    setSlashIndex(prev => (prev + 1) % filteredCmds.length);
                                    return;
                                }
                                if (e.key === 'ArrowUp') {
                                    e.preventDefault();
                                    setSlashIndex(prev => (prev - 1 + filteredCmds.length) % filteredCmds.length);

                        onBlur={e => {
